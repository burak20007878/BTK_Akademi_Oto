// ==UserScript==
// @name         BTK Akademi Arka Plan Tam Otomasyon (1.5x + Tik + Oto Bolum)
// @match        *://*.btkakademi.gov.tr/*
// @match        *://*.cinema8.com/*
// @allFrames    true
// @run-at       document-start
// ==/UserScript==

(() => {
  const TARGET_SPEED = 4;
  const IS_TOP = window.self === window.top;

  // 1. Görünürlük, sekme gizliliği ve odak kontrollerini devre dışı bırak
  const unlockVisibility = (win) => {
    try {
      if (!win || win._btkVisUnlocked) return;
      win._btkVisUnlocked = true;
      const doc = win.document;

      Object.defineProperty(doc, 'hidden', { get: () => false, configurable: true });
      Object.defineProperty(doc, 'visibilityState', { get: () => 'visible', configurable: true });
      try {
        Object.defineProperty(doc, 'hasFocus', { value: () => true, configurable: true });
      } catch (e) {}

      ['visibilitychange', 'webkitvisibilitychange', 'mozvisibilitychange', 'blur', 'pagehide', 'focusout'].forEach(ev => {
        win.addEventListener(ev, e => e.stopImmediatePropagation(), true);
        doc.addEventListener(ev, e => e.stopImmediatePropagation(), true);
      });
    } catch (e) {}
  };

  unlockVisibility(window);

  // 2. Arka plan uyku modunu engelle (duyulmayan Web Audio sinyali)
  const keepTabAwake = () => {
    let started = false;
    const initAudio = () => {
      if (started) return;
      try {
        const AudioCtx = window.AudioContext || window.webkitAudioContext;
        if (!AudioCtx) return;
        const ctx = new AudioCtx();
        const osc = ctx.createOscillator();
        const gain = ctx.createGain();
        gain.gain.value = 0.00001;
        osc.connect(gain);
        gain.connect(ctx.destination);
        osc.start();
        if (ctx.state === 'suspended') ctx.resume();
        started = true;
      } catch (e) {}
    };
    initAudio();
    ['click', 'pointerdown', 'keydown'].forEach(ev => {
      window.addEventListener(ev, initAudio, { capture: true, passive: true });
    });
  };

  keepTabAwake();

  // 3. Arka planda yavaşlatılmayan Web Worker zamanlayıcısı
  const createUnthrottledInterval = (fn, ms) => {
    try {
      const blob = new Blob([`setInterval(() => postMessage(1), ${ms});`], { type: 'application/javascript' });
      const worker = new Worker(URL.createObjectURL(blob));
      worker.onmessage = fn;
      return worker;
    } catch (e) {
      return setInterval(fn, ms);
    }
  };

  // 4. Arka planda dondurulan requestAnimationFrame ve IntersectionObserver'ı açık tut
  const origRAF = window.requestAnimationFrame.bind(window);
  window.requestAnimationFrame = (cb) => {
    let fired = false;
    const timer = setTimeout(() => {
      if (!fired) {
        fired = true;
        cb(performance.now());
      }
    }, 35);
    return origRAF((ts) => {
      if (!fired) {
        fired = true;
        clearTimeout(timer);
        cb(ts);
      }
    });
  };

  if (window.IntersectionObserver) {
    const OrigIO = window.IntersectionObserver;
    window.IntersectionObserver = function (callback, options) {
      const wrappedCb = (entries, observer) => {
        const fakeEntries = entries.map(entry => new Proxy(entry, {
          get(target, prop) {
            if (prop === 'isIntersecting') return true;
            if (prop === 'intersectionRatio') return 1;
            const val = Reflect.get(target, prop, target);
            return typeof val === 'function' ? val.bind(target) : val;
          }
        }));
        callback(fakeEntries, observer);
      };
      return new OrigIO(wrappedCb, options);
    };
    window.IntersectionObserver.prototype = OrigIO.prototype;
  }

  // 5. Sesli otomatik oynatma engellenirse otomatik sessize alıp başlat
  const origPlay = HTMLMediaElement.prototype.play;
  HTMLMediaElement.prototype.play = function (...args) {
    const p = origPlay.apply(this, args);
    if (p && typeof p.catch === 'function') {
      return p.catch(() => {
        this.muted = true;
        this.defaultMuted = true;
        this.setAttribute('muted', '');
        return origPlay.apply(this, args);
      });
    }
    return p;
  };

  // 6. ÜST SAYFA (btkakademi.gov.tr): Tik takibi + kapalı bölüm açma + sonraki video
  if (IS_TOP) {
    let lastSwitchTime = 0;
    let lastExpandTime = 0;
    let currentLessonKey = '';
    let lessonEnteredAt = 0;
    let expectedDurationSec = 300;
    let initialIconShape = '';
    const knownTickShapes = new Set();

    const triggerClick = (el) => {
      if (!el) return;
      let target = el.querySelector('a, button, [role="button"]');
      if (!target) {
        let cur = el;
        for (let i = 0; i < 4 && cur && cur !== document.body; i++) {
          if (cur.tagName === 'LI' || cur.getAttribute('role') === 'button' || getComputedStyle(cur).cursor === 'pointer') {
            target = cur;
            break;
          }
          cur = cur.parentElement;
        }
      }
      target = target || el;
      target.dispatchEvent(new MouseEvent('mousedown', { bubbles: true, cancelable: true }));
      target.dispatchEvent(new MouseEvent('mouseup', { bubbles: true, cancelable: true }));
      target.click();
    };

    const parseDurationSeconds = (text) => {
      let total = 0;
      const h = text.match(/(\d+)\s*sa/i);
      const m = text.match(/(\d+)\s*(?:d|dk)(?!\w)/i);
      const s = text.match(/(\d+)\s*sn/i);
      if (h) total += parseInt(h[1], 10) * 3600;
      if (m) total += parseInt(m[1], 10) * 60;
      if (s) total += parseInt(s[1], 10);
      return total || 240;
    };

    // Bölüm başlıklarını ("( 0/7 )", "( 4/8 )") ve açık ders satırlarını ayrı ayrı bul
    const scanSidebar = () => {
      const durationNodes = Array.from(document.querySelectorAll('span, div, p, small')).filter(el => {
        if (el.children.length > 0) return false;
        const txt = (el.textContent || '').trim();
        return /^\d+\s*(sa|d|dk|sn)(\s+\d+\s*(d|dk|sn))?$/i.test(txt);
      });

      const lessons = [];
      const headers = [];
      const seen = new Set();

      for (const durEl of durationNodes) {
        let row = durEl.parentElement;
        for (let i = 0; i < 4 && row; i++) {
          const hasIcon = row.querySelector('svg, i');
          const fullText = (row.textContent || '').trim();
          const durText = (durEl.textContent || '').trim();
          if (hasIcon && fullText.length > durText.length + 3 && row.offsetHeight > 20 && row.offsetHeight < 110) {
            break;
          }
          row = row.parentElement;
        }
        if (!row || seen.has(row)) continue;
        seen.add(row);

        const fullText = (row.textContent || '').trim();
        const countMatch = fullText.match(/\(\s*(\d+)\s*\/\s*(\d+)\s*\)/);

        if (countMatch) {
          headers.push({
            el: row,
            done: parseInt(countMatch[1], 10),
            total: parseInt(countMatch[2], 10),
            top: row.getBoundingClientRect().top
          });
        } else {
          lessons.push(row);
        }
      }
      return { lessons, headers };
    };

    const parseRgb = (str) => {
      const m = (str || '').match(/rgba?\((\d+),\s*(\d+),\s*(\d+)/);
      return m ? [+m[1], +m[2], +m[3]] : null;
    };

    const isBlue = (str) => {
      const rgb = parseRgb(str);
      return rgb ? (rgb[2] > 140 && (rgb[2] - rgb[0]) > 50) : false;
    };

    const isGreen = (str) => {
      const rgb = parseRgb(str);
      return rgb ? (rgb[1] > 120 && (rgb[1] - rgb[0]) > 35 && (rgb[1] - rgb[2]) > 20) : false;
    };

    const getIconShape = (row) => {
      const icon = row.querySelector('svg, i');
      if (!icon) return '';
      if (icon.tagName.toLowerCase() === 'svg') {
        const paths = Array.from(icon.querySelectorAll('path, polyline, polygon, circle'))
          .map(p => p.getAttribute('d') || p.getAttribute('points') || '')
          .join('|');
        return `svg:${icon.getAttribute('data-icon') || ''}:${paths}`;
      }
      return `i:${icon.className}`;
    };

    const learnTickShapes = (rows) => {
      for (const row of rows) {
        const icon = row.querySelector('svg, i');
        if (!icon) continue;
        const pathEl = icon.querySelector('path, polyline, circle');
        const colors = [
          getComputedStyle(icon).color,
          getComputedStyle(icon).fill,
          pathEl ? getComputedStyle(pathEl).fill : '',
          pathEl ? getComputedStyle(pathEl).stroke : ''
        ];
        if (colors.some(isGreen)) {
          const shape = getIconShape(row);
          if (shape) knownTickShapes.add(shape);
        }
      }
    };

    const isRowActive = (row) => {
      if (row.getAttribute('aria-current') === 'true' || /active|selected/i.test(row.className || '')) return true;
      const els = [row, ...row.querySelectorAll('span, div, p, a')];
      return els.some(el => isBlue(getComputedStyle(el).color));
    };

    const isRowTicked = (row) => {
      const shape = getIconShape(row);
      if (!shape) return false;
      if (knownTickShapes.has(shape)) return true;

      const icon = row.querySelector('svg, i');
      const pathEl = icon?.querySelector('path, polyline, circle');
      const colors = [
        icon ? getComputedStyle(icon).color : '',
        pathEl ? getComputedStyle(pathEl).fill : '',
        pathEl ? getComputedStyle(pathEl).stroke : ''
      ];
      return colors.some(isGreen);
    };

    // Kapalı duran sıradaki bölüm başlığını bul ve aç
    const expandNextIncompleteSection = (headers, afterTop = -Infinity) => {
      const now = Date.now();
      if (now - lastExpandTime < 2500) return false;

      const candidate = headers.find(h => h.done < h.total && h.top > afterTop + 5);
      if (candidate) {
        lastExpandTime = now;
        console.log('[BTK Oto] Kapalı bölüm açılıyor:', (candidate.el.textContent || '').trim().slice(0, 40));
        triggerClick(candidate.el);
        return true;
      }
      return false;
    };

    const goNextLesson = (lessons, headers, activeIndex) => {
      const now = Date.now();
      if (now - lastSwitchTime < 3000) return;

      // 1. Açık listede bir sonraki ders varsa doğrudan ona tıkla
      if (activeIndex !== -1 && activeIndex + 1 < lessons.length) {
        lastSwitchTime = now;
        const nextRow = lessons[activeIndex + 1];
        console.log('[BTK Oto] Sonraki videoya geçiliyor:', (nextRow.textContent || '').trim().slice(0, 45));
        triggerClick(nextRow);
        return;
      }

      // 2. Açık bölümün son videosundaysak alttaki kapalı bölümü aç
      const activeTop = activeIndex !== -1 ? lessons[activeIndex].getBoundingClientRect().top : -Infinity;
      if (expandNextIncompleteSection(headers, activeTop)) {
        return;
      }

      // 3. Yedek: Sayfadaki "Sonraki" butonunu dene
      const buttons = Array.from(document.querySelectorAll('button, a, [role="button"]'));
      const nextBtn = buttons.find(el => {
        const t = (el.innerText || el.getAttribute('aria-label') || el.title || '').trim().toLowerCase();
        return t === 'sonraki' || t === 'sonraki ders' || t === 'sonraki video' || t === 'ileri' || t.includes('sonraki derse');
      });
      if (nextBtn && !nextBtn.disabled) {
        lastSwitchTime = now;
        triggerClick(nextBtn);
      }
    };

    const checkAndAdvance = () => {
      unlockVisibility(window);

      document.querySelectorAll('iframe').forEach(ifr => {
        if (ifr.getAttribute('loading') === 'lazy') ifr.setAttribute('loading', 'eager');
        const allow = ifr.getAttribute('allow') || '';
        if (!allow.includes('autoplay')) {
          ifr.setAttribute('allow', `${allow}; autoplay; fullscreen; encrypted-media`.replace(/^;\s*/, ''));
        }
      });

      const { lessons, headers } = scanSidebar();
      if (!lessons.length && !headers.length) return;

      learnTickShapes(lessons);

      const activeIndex = lessons.findIndex(isRowActive);

      // Aktif ders kapalı bir bölümün içinde kaldıysa o bölümü aç
      if (activeIndex === -1) {
        expandNextIncompleteSection(headers, -Infinity);
        return;
      }

      const activeRow = lessons[activeIndex];
      const lessonKey = (activeRow.textContent || '').trim();
      const shapeNow = getIconShape(activeRow);

      if (lessonKey !== currentLessonKey) {
        currentLessonKey = lessonKey;
        lessonEnteredAt = Date.now();
        expectedDurationSec = parseDurationSeconds(lessonKey);
        initialIconShape = shapeNow;
        console.log(`[BTK Oto] Aktif ders: ${lessonKey.slice(0, 40)} (Tahmini süre: ${Math.round(expectedDurationSec / TARGET_SPEED)} sn)`);
        return;
      }

      const elapsedSec = (Date.now() - lessonEnteredAt) / 1000;
      if (elapsedSec < 3) return;

      const shapeChanged = initialIconShape && shapeNow && shapeNow !== initialIconShape;
      const tickedNow = isRowTicked(activeRow);
      const nextIsUnticked = activeIndex + 1 < lessons.length && !isRowTicked(lessons[activeIndex + 1]);
      // Video süresi 1.5x hızda dolduğu halde +30 sn geçtiyse takılmayı önlemek için zorla geç
      const timedOut = elapsedSec > (expectedDurationSec / TARGET_SPEED) + 30;

      if ((shapeChanged && tickedNow) || (tickedNow && nextIsUnticked) || timedOut) {
        if (timedOut) console.log('[BTK Oto] Süre aşımı koruması devreye girdi, sonraki derse geçiliyor...');
        goNextLesson(lessons, headers, activeIndex);
      }
    };

    createUnthrottledInterval(checkAndAdvance, 800);

    window.addEventListener('message', e => {
      if (e.data === 'BTK_NEXT_VIDEO') {
        const { lessons, headers } = scanSidebar();
        const activeIndex = lessons.findIndex(isRowActive);
        goNextLesson(lessons, headers, activeIndex);
      }
    });

    return;
  }

  // 7. VİDEO ÇERÇEVESİ (cinema8.com): Gerçek ilerleme takibi + arayüz oynatma tetikleyici + 1.5x hız
  let lastActionTime = 0;
  let lastObservedTime = -1;
  let lastProgressAt = Date.now();

  const keepPlayingInBackground = () => {
    unlockVisibility(window);

    const video = document.querySelector('video');
    if (!video) return;

    if (!video._btkHooked) {
      video._btkHooked = true;
      video.muted = true;
      video.autoplay = true;
      video.setAttribute('muted', '');
      video.setAttribute('autoplay', '');

      video.addEventListener('ratechange', () => {
        if (video.playbackRate !== TARGET_SPEED) video.playbackRate = TARGET_SPEED;
      });
      video.addEventListener('ended', () => {
        console.log('[BTK Oto] Video sonuna ulaşıldı.');
        window.top.postMessage('BTK_NEXT_VIDEO', '*');
      });
    }

    if (video.playbackRate !== TARGET_SPEED) {
      video.playbackRate = TARGET_SPEED;
    }

    // Videonun gerçekten saniye sayıp saymadığını kontrol et
    if (Math.abs(video.currentTime - lastObservedTime) > 0.15) {
      lastObservedTime = video.currentTime;
      lastProgressAt = Date.now();
    }

    const isStalled = (Date.now() - lastProgressAt) > 3500;
    const now = Date.now();

    if ((video.paused || isStalled) && !video.ended && (now - lastActionTime > 1200)) {
      lastActionTime = now;

      // 1. Önce Cinema8 / Video.js arayüz butonlarını tıkla (sayaç senkronizasyonu için)
      const playSelectors = [
        '.vjs-big-play-button',
        '.vjs-play-control.vjs-paused',
        'button[title*="Play" i]',
        'button[title*="Oynat" i]',
        '[class*="big-play"]'
      ];
      for (const sel of playSelectors) {
        const btn = document.querySelector(sel);
        if (btn && btn.offsetParent !== null) {
          btn.click();
          break;
        }
      }

      // 2. Video.js API üzerinden başlat
      if (window.videojs?.getAllPlayers) {
        window.videojs.getAllPlayers().forEach(p => {
          p.muted(true);
          if (p.playbackRate() !== TARGET_SPEED) p.playbackRate(TARGET_SPEED);
          if (p.paused()) p.play();
        });
      }

      // 3. Doğrudan <video> elementini oynat
      video.muted = true;
      video.play();
    }
  };

  createUnthrottledInterval(keepPlayingInBackground, 700);
})();
